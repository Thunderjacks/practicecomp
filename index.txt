<script>
/* =========================================================
   SETUP: paste your Google Apps Script web app URL below.
   It looks like https://script.google.com/macros/s/XXXX/exec
   The exec PIN now lives in the Apps Script, not in this file.
   ========================================================= */
const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxSiZm9xnV3Ie9BUAry_Px9x0aZlYxxtwhDY6WGWCUvEZdZlMpoE2RA2xIRBDJLnqYN/exec";

const ALL_EVENTS = ["Underhand Chop","Pole Climb","Stock Saw","Axe Throwing","Single Buck","Double Buck","Jack & Jill","Obstacle Pole","Choker Race","Speed Axe Throw","Vertical Chop"];
const POLL_MS = 20000;

let execMode = false;
let execPin = null;
let comp = { title:"", events:[], directions:{} };
let checkins = [];
let scores = {};
let sending = false;

function configured(){ return SCRIPT_URL.indexOf('https://script.google.com/') === 0; }

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));
}

function showMsg(text, type){
  const slot = document.getElementById('msgSlot');
  slot.innerHTML = '<div class="msg ' + type + '">' + escapeHtml(text) + '</div>';
  clearTimeout(showMsg._t);
  showMsg._t = setTimeout(() => { slot.innerHTML = ''; }, 5000);
}

/* ---------------- server calls ---------------- */
async function apiGet(){
  const r = await fetch(SCRIPT_URL);
  if(!r.ok) throw new Error('HTTP ' + r.status);
  return r.json();
}
async function apiPost(body){
  // text/plain body keeps this a "simple" request, which avoids CORS preflight problems with Apps Script
  const r = await fetch(SCRIPT_URL, { method:'POST', body: JSON.stringify(body) });
  if(!r.ok) throw new Error('HTTP ' + r.status);
  return r.json();
}
function applyState(s){
  if(!s) return;
  comp = s.comp || comp;
  if(!comp.directions) comp.directions = {};
  checkins = s.checkins || [];
  scores = s.scores || {};
}
async function send(body, okMsg){
  try{
    const res = await apiPost(body);
    if(!res.ok){
      if(res.error === 'bad pin'){
        execMode = false; execPin = null;
        showMsg('Exec session expired. Unlock again.', 'err');
        render(); return null;
      }
      showMsg(res.error || 'Something went wrong.', 'err');
      return null;
    }
    if(res.state) applyState(res.state);
    if(okMsg) showMsg(okMsg, 'ok');
    return res;
  }catch(e){
    showMsg("Couldn't reach the server. Check your connection and try again.", 'err');
    return null;
  }
}
async function loadAll(){
  if(!configured()){
    document.getElementById('compSub').textContent = 'Not connected';
    showMsg('This page is not connected to its Google Sheet yet (SCRIPT_URL not set).', 'err');
    return;
  }
  try{
    const res = await apiGet();
    if(res.ok) applyState(res.state);
    render();
  }catch(e){
    document.getElementById('compSub').textContent = 'Offline';
    showMsg("Couldn't load the latest data. Retrying automatically.", 'err');
  }
}

/* ---------------- rendering ---------------- */
function render(){
  document.getElementById('compSub').textContent = comp.title
    ? comp.title + (comp.events.length ? ' · ' + comp.events.length + ' events' : '')
    : 'No active comp set up yet';
  document.getElementById('execBadgeSlot').innerHTML = execMode ? '<span class="exec-badge">Exec mode</span>' : '';
  document.getElementById('setupCard').style.display = execMode ? 'block' : 'none';
  document.getElementById('execGateCard').style.display = execMode ? 'none' : 'block';
  renderCheckinEvents();
  renderRoster();
  renderScoreboard();
}

function renderSetup(){
  document.getElementById('compTitle').value = comp.title || '';
  document.getElementById('eventCheckboxes').innerHTML = ALL_EVENTS.map(ev => {
    const checked = comp.events.includes(ev) ? 'checked' : '';
    return '<label class="ev-check"><input type="checkbox" value="' + escapeHtml(ev) + '" ' + checked + '> ' + escapeHtml(ev) + '</label>';
  }).join('');
}

function renderCheckinEvents(){
  const grid = document.getElementById('checkinEvents');
  const prev = new Set(Array.from(grid.querySelectorAll('input:checked')).map(i => i.value));
  const btn = document.getElementById('checkinBtn');
  if(comp.events.length === 0){
    grid.innerHTML = '<p class="empty">No comp set up yet. Check back once an exec starts one.</p>';
    btn.disabled = true;
  }else{
    btn.disabled = sending;
    grid.innerHTML = comp.events.map(ev =>
      '<label class="ev-check"><input type="checkbox" value="' + escapeHtml(ev) + '"' + (prev.has(ev) ? ' checked' : '') + '> ' + escapeHtml(ev) + '</label>'
    ).join('');
  }
}

function renderRoster(){
  const el = document.getElementById('rosterList');
  if(checkins.length === 0){
    el.innerHTML = '<p class="empty">No one checked in yet.</p>';
    return;
  }
  el.innerHTML = '<table><tr><th>Name</th><th>Events</th>' + (execMode ? '<th></th>' : '') + '</tr>' +
    checkins.map(c =>
      '<tr><td>' + escapeHtml(c.name) + '</td><td>' + c.events.map(escapeHtml).join(', ') + '</td>' +
      (execMode ? '<td><button class="small secondary" data-remove="' + escapeHtml(c.id) + '">Remove</button></td>' : '') +
      '</tr>'
    ).join('') + '</table>';
  if(execMode){
    el.querySelectorAll('[data-remove]').forEach(b => b.addEventListener('click', async () => {
      if(!confirm('Remove this check-in (and any scores entered for it)?')) return;
      await send({ action:'removeCheckin', pin:execPin, id:b.dataset.remove }, 'Removed.');
      render();
    }));
  }
}

function renderScoreboard(){
  const el = document.getElementById('scoreboardList');
  if(comp.events.length === 0){
    el.innerHTML = '<p class="empty">Nothing to show yet.</p>';
    return;
  }
  let html = '';
  const overall = {};
  const pointsForRank = [10,8,6,5,4,3,2,1];

  comp.events.forEach(ev => {
    const participants = checkins.filter(c => c.events.includes(ev));
    if(participants.length === 0) return;
    const dir = (comp.directions && comp.directions[ev]) || 'desc';
    const evScores = scores[ev] || {};
    const has = v => v !== undefined && v !== null && v !== '';
    const rows = participants.map(p => ({ id:p.id, name:p.name, score:evScores[p.id] }));
    const scored = rows.filter(r => has(r.score)).sort((a,b) => dir === 'asc' ? a.score - b.score : b.score - a.score);
    const unscored = rows.filter(r => !has(r.score));
    scored.forEach((r,i) => { overall[r.name] = (overall[r.name] || 0) + (pointsForRank[i] || 0); });

    const encEv = encodeURIComponent(ev);
    html += '<div class="event-block"><h3>' + escapeHtml(ev) + ' <span style="font-family:var(--mono); font-size:.65rem; color:var(--bark);">(' + (dir === 'asc' ? 'lowest wins' : 'highest wins') + ')</span></h3>';
    scored.forEach((r,i) => {
      html += '<div class="score-row"><span class="name ' + (i === 0 ? 'rank1' : '') + '">' + (i+1) + '. ' + escapeHtml(r.name) + '</span>';
      html += execMode
        ? '<input type="number" step="0.01" class="score-input" data-event="' + encEv + '" data-id="' + escapeHtml(r.id) + '" value="' + r.score + '">'
        : '<span style="font-family:var(--mono);">' + r.score + '</span>';
      html += '</div>';
    });
    unscored.forEach(r => {
      html += '<div class="score-row"><span class="name">' + escapeHtml(r.name) + '</span>';
      html += execMode
        ? '<input type="number" step="0.01" class="score-input" placeholder="score" data-event="' + encEv + '" data-id="' + escapeHtml(r.id) + '" value="">'
        : '<span style="color:var(--bark); font-family:var(--mono); font-size:.8rem;">pending</span>';
      html += '</div>';
    });
    if(execMode){
      html += '<div style="margin-top:8px; display:flex; gap:8px; align-items:center;"><label style="margin:0;">Sort:</label>' +
        '<select class="dir-select" data-event="' + encEv + '" style="font-family:var(--mono); font-size:.72rem; padding:4px;">' +
        '<option value="desc"' + (dir === 'desc' ? ' selected' : '') + '>Highest wins</option>' +
        '<option value="asc"' + (dir === 'asc' ? ' selected' : '') + '>Lowest wins (time)</option></select></div>';
    }
    html += '</div>';
  });

  const list = Object.entries(overall).sort((a,b) => b[1] - a[1]);
  if(list.length){
    html += '<div class="event-block"><h3 style="color:var(--moss);">Overall standings</h3><table><tr><th>Name</th><th>Points</th></tr>' +
      list.map(([n,p],i) => '<tr><td class="' + (i === 0 ? 'rank1' : '') + '">' + escapeHtml(n) + '</td><td>' + p + '</td></tr>').join('') + '</table></div>';
  }
  el.innerHTML = html || '<p class="empty">No one has checked in for the active events yet.</p>';

  if(execMode){
    el.querySelectorAll('.score-input').forEach(inp => inp.addEventListener('change', async e => {
      const ev = decodeURIComponent(e.target.dataset.event);
      const val = e.target.value === '' ? null : parseFloat(e.target.value);
      await send({ action:'setScore', pin:execPin, event:ev, id:e.target.dataset.id, value:val });
      renderScoreboard();
    }));
    el.querySelectorAll('.dir-select').forEach(sel => sel.addEventListener('change', async e => {
      const ev = decodeURIComponent(e.target.dataset.event);
      await send({ action:'setDirection', pin:execPin, event:ev, dir:e.target.value });
      renderScoreboard();
    }));
  }
}

/* ---------------- actions ---------------- */
document.getElementById('pinBtn').addEventListener('click', async () => {
  const pin = document.getElementById('pinInput').value;
  if(!pin) return;
  if(!configured()){ showMsg('Not connected to the Google Sheet yet.', 'err'); return; }
  try{
    const res = await apiPost({ action:'verify', pin });
    if(res.ok){
      execMode = true; execPin = pin;
      document.getElementById('pinInput').value = '';
      showMsg('Exec mode unlocked.', 'ok');
      render(); renderSetup();
    }else{
      showMsg('Wrong PIN.', 'err');
    }
  }catch(e){
    showMsg("Couldn't reach the server.", 'err');
  }
});

document.getElementById('lockBtn').addEventListener('click', () => {
  execMode = false; execPin = null; render();
});

document.getElementById('startCompBtn').addEventListener('click', async () => {
  const title = document.getElementById('compTitle').value.trim();
  if(!title){ showMsg('Enter a comp title first.', 'err'); return; }
  const events = Array.from(document.querySelectorAll('#eventCheckboxes input:checked')).map(i => i.value);
  if(events.length === 0){ showMsg('Select at least one event.', 'err'); return; }
  const directions = Object.assign({}, comp.directions);
  events.forEach(ev => { if(!directions[ev]) directions[ev] = 'desc'; });
  const res = await send({ action:'setComp', pin:execPin, comp:{ title, events, directions } }, 'Comp updated.');
  if(res){ render(); renderSetup(); }
});

document.getElementById('resetCompBtn').addEventListener('click', async () => {
  if(!confirm('Clear all current check-ins and scores? This cannot be undone.')) return;
  const res = await send({ action:'reset', pin:execPin }, 'Check-ins and scores cleared.');
  if(res) render();
});

document.getElementById('checkinBtn').addEventListener('click', async () => {
  if(sending) return;
  const nameEl = document.getElementById('nameInput');
  const name = nameEl.value.trim();
  if(!name){ showMsg('Enter your name first.', 'err'); return; }
  const events = Array.from(document.querySelectorAll('#checkinEvents input:checked')).map(i => i.value);
  if(events.length === 0){ showMsg('Pick at least one event.', 'err'); return; }
  sending = true;
  document.getElementById('checkinBtn').disabled = true;
  const res = await send({ action:'checkin', name, events }, 'Checked in! Good luck out there.');
  sending = false;
  if(res){
    nameEl.value = '';
    document.querySelectorAll('#checkinEvents input:checked').forEach(i => i.checked = false);
  }
  render();
});

/* ---------------- live refresh ---------------- */
setInterval(() => {
  if(document.hidden || !configured()) return;
  const a = document.activeElement;
  if(a && (a.tagName === 'INPUT' || a.tagName === 'SELECT' || a.tagName === 'TEXTAREA')) return;
  loadAll();
}, POLL_MS);

loadAll();
</script>
